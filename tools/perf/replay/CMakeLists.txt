cmake_minimum_required(VERSION 3.16)
project(replay C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(project_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(root "${project_root}/external/openh264")
file(GLOB common "${root}/codec/common/src/*.cpp")
file(GLOB decoder "${root}/codec/decoder/core/src/*.cpp" "${root}/codec/decoder/plus/src/*.cpp")
list(REMOVE_ITEM decoder "${root}/codec/decoder/core/src/decoder_core.cpp" "${root}/codec/decoder/plus/src/welsDecoderExt.cpp")
set(patches "${project_root}/build/esp-idf/openh264_sw")
if(FAST)
  list(REMOVE_ITEM decoder "${root}/codec/decoder/core/src/cabac_decoder.cpp" "${root}/codec/decoder/core/src/parse_mb_syn_cabac.cpp")
  list(APPEND decoder "${project_root}/components/openh264_sw/cabac_decoder_esp.cpp" "${project_root}/build/esp-idf/openh264_sw/parse_mb_syn_cabac_esp.cpp")
endif()
add_library(openh264 STATIC ${common} ${decoder} "${patches}/decoder_core_esp.cpp" "${patches}/welsDecoderExt_esp.cpp")
target_include_directories(openh264 PUBLIC "${root}/codec/api" "${root}/codec/api/wels" PRIVATE "${root}/codec/common/inc" "${root}/codec/decoder/core/inc" "${root}/codec/decoder/plus/inc" "${project_root}/components/openh264_sw")
target_compile_options(openh264 PRIVATE -O2 -fno-strict-aliasing -w)
add_executable(replay replay.cpp "${project_root}/main/core/video_decoder.cpp" "${project_root}/main/core/pixel_convert.c")
target_include_directories(replay PRIVATE "${project_root}/main")
target_link_libraries(replay PRIVATE openh264 pthread)
add_executable(crc crc.cpp "${project_root}/main/core/video_decoder.cpp" "${project_root}/main/core/pixel_convert.c")
target_include_directories(crc PRIVATE "${project_root}/main")
target_link_libraries(crc PRIVATE openh264 pthread z)
