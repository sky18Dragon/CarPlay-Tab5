option(TAB5_SOFTWARE_AUTH "Embed locally supplied experimental iAP2 identity" OFF)
if(TAB5_SOFTWARE_AUTH)
    set(identity_dir "${CMAKE_CURRENT_LIST_DIR}/../../secrets/mfi")
    foreach(asset identity.pk8 certificate.p7b certificate.der)
        if(NOT EXISTS "${identity_dir}/${asset}")
            message(FATAL_ERROR "Software authentication requires secrets/mfi/${asset}")
        endif()
    endforeach()
    idf_component_register(SRCS "auth_software_provider.c" INCLUDE_DIRS "include" REQUIRES auth_port mbedtls esp_hw_support freertos)
    foreach(asset identity.pk8 certificate.p7b certificate.der)
        target_add_binary_data(${COMPONENT_LIB} "${identity_dir}/${asset}" BINARY)
    endforeach()
    target_compile_definitions(${COMPONENT_LIB} INTERFACE TAB5_SOFTWARE_AUTH=1)
else()
    idf_component_register(INCLUDE_DIRS "include" REQUIRES auth_port mbedtls esp_hw_support freertos)
endif()
