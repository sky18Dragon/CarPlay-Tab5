cmake_minimum_required(VERSION 3.16)
project(tab5_ui_render C CXX)
get_filename_component(root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${root}/managed_components/lvgl__lvgl" lvgl)
add_executable(render render.c "${root}/main/app_ui.c" "${root}/main/core/touch.c" "${root}/main/core/session.c" "${root}/main/core/pixel_convert.c")
target_include_directories(render PRIVATE stubs "${root}/main" "${root}/components/airplay_control/include")
target_compile_options(render PRIVATE -Wall -Wextra -Werror -UNDEBUG)
target_link_libraries(render PRIVATE lvgl m)
